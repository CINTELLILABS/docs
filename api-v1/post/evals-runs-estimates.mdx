---
title: "Estimate Eval Run"
api: "POST https://api.bland.ai/v1/evals/runs/estimates"
description: "Preview the size and cost of a run before starting it."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Body Parameters

Accepts the same body as [Create Eval Run](/api-v1/post/evals-runs):

<ParamField body="call_ids" type="array of string" required>
  The call IDs to include in the estimate. Between 1 and 5000 calls.
</ParamField>

<ParamField body="attached_agents" type="array of objects">
  The eval agents to score with. Maximum 50 agents. Required when `workbench_setup_version_id` is not provided.

  <Expandable title="Agent attachment fields">
    <ParamField body="eval_agent_id" type="string" required>
      The UUID of the eval agent.
    </ParamField>

    <ParamField body="eval_agent_version_id" type="string">
      Pin a specific version of the agent. Omit to use the latest published version.
    </ParamField>

    <ParamField body="weight" type="number">
      Relative weight of this agent in the overall score, 0-100.
    </ParamField>

    <ParamField body="target_level_keys" type="array of string">
      Level keys considered a passing result for this agent.
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="run_mode" type="string">
  The scoring modality. One of `text`, `audio`, or `full`. Defaults to `text`.
</ParamField>

<ParamField body="pass_threshold_pct" type="number or null">
  Percentage of evaluations that must pass for the run to be marked passing overall, 0-100. Omit this field or set it to `null` to estimate a run without a threshold.
</ParamField>

<ParamField body="workbench_setup_id" type="string">
  UUID of a workbench setup to associate with the run. Must be provided together with `workbench_setup_version_id`.
</ParamField>

<ParamField body="workbench_setup_version_id" type="string">
  Pinned workbench setup version whose agent roster supplies the agents. Required when `attached_agents` is not provided. Must be provided together with `workbench_setup_id`.
</ParamField>

<ParamField body="agent_id" type="string">
  UUID of the voice agent the estimated run evaluates. This identifies the voice agent, not an eval agent used as a judge.
</ParamField>

<ParamField body="agent_version_id" type="string">
  UUID of the specific voice agent version the estimated run evaluates.
</ParamField>

<ParamField body="async" type="boolean" default="true">
  Whether judging may use delay-tolerant execution lanes. Set this to `false` to estimate synchronous judging within the background run workflow.
</ParamField>

<ParamField body="triggered_by" type="string">
  How the run is initiated. One of `manual`, `auto`, or `backfill`. Defaults to `manual`.
</ParamField>

<ParamField body="metadata" type="object">
  Key-value metadata to attach to the run. Keys and values must be strings.
</ParamField>

<Note>
  You must supply either a non-empty `attached_agents` array or a `workbench_setup_version_id` whose pinned roster supplies the agents. If you provide `workbench_setup_id` or `workbench_setup_version_id`, both fields are required together. Unknown body fields return `400`.
</Note>

### Response

<ResponseField name="resolved_atom_count" type="integer">
  Number of call-by-agent evaluations the run will perform.
</ResponseField>

<ResponseField name="estimated_input_tokens" type="integer">
  Estimated number of input tokens consumed by judge calls.
</ResponseField>

<ResponseField name="estimated_output_tokens" type="integer">
  Estimated number of output tokens produced by judge calls.
</ResponseField>

<ResponseField name="estimated_cost_usd_cents" type="number">
  Estimated billable cost in USD cents.
</ResponseField>

<ResponseField name="estimated_baseline_cost_usd_cents" type="number">
  Estimated customer cost in USD cents if the same prompts used uncached synchronous execution.
</ResponseField>

<ResponseField name="estimated_savings_pct" type="number">
  Estimated savings percentage against the uncached synchronous baseline, before margin.
</ResponseField>

<ResponseField name="execution_lanes" type="array of objects">
  Cost-aware execution lanes selected for the estimated run.

  <Expandable title="Execution lane fields">
    <ResponseField name="lane" type="string">
      Execution lane. One of `sync_uncached`, `sync_explicit_cache`, or `batch`.
    </ResponseField>

    <ResponseField name="atom_count" type="integer">
      Number of call-by-agent evaluations assigned to this lane.
    </ResponseField>
  </Expandable>
</ResponseField>
