---
title: "Update Workbench Setup"
api: "PATCH https://api.bland.ai/v1/evals/workbench-setups/{setup_id}"
description: "Update a workbench setup's metadata or repoint its active version."
---

<Note>
  This endpoint updates setup-level fields only. To edit attached agents, the pass threshold, or run mode, update the setup's draft version using the Update Workbench Setup Version endpoint.
</Note>

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="setup_id" type="string" required>
  The ID of the workbench setup to update.
</ParamField>

### Body Parameters

All fields are optional. Only include fields you want to update.

<ParamField body="name" type="string">
  Updated display name for the setup. Between 1 and 200 characters.
</ParamField>

<ParamField body="description" type="string | null">
  Updated description, or `null` to clear it.
</ParamField>

<ParamField body="active_version_id" type="string | null">
  Repoint the published version to a different archived version ID. Set to `null` to unpublish the setup.
</ParamField>

<ParamField body="metadata" type="object">
  Key-value metadata to associate with the setup. Values must be strings.
</ParamField>

### Response

<ResponseField name="data" type="object">
  The updated workbench setup.

  <Expandable title="setup fields">
    <ResponseField name="id" type="string">
      Unique identifier for the workbench setup.
    </ResponseField>
    <ResponseField name="org_id" type="string">
      ID of the organization that owns this setup.
    </ResponseField>
    <ResponseField name="key" type="string">
      Stable slug key for the setup.
    </ResponseField>
    <ResponseField name="name" type="string">
      Display name of the setup.
    </ResponseField>
    <ResponseField name="description" type="string | null">
      Description of the setup, or `null` if not set.
    </ResponseField>
    <ResponseField name="current_version_id" type="string">
      ID of the current editable draft version.
    </ResponseField>
    <ResponseField name="active_version_id" type="string | null">
      ID of the published version, or `null` if unpublished.
    </ResponseField>
    <ResponseField name="agent_id" type="string | null">
      UUID of the associated voice agent, or `null` for an organization-wide setup.
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Key-value metadata. Values are strings.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp for when the setup was created.
    </ResponseField>
    <ResponseField name="updated_at" type="string">
      ISO 8601 timestamp for when the setup was last updated.
    </ResponseField>
    <ResponseField name="deleted_at" type="string | null">
      ISO 8601 timestamp for when the setup was deleted, or `null` if not deleted.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="null">
  `null` on success.
</ResponseField>
