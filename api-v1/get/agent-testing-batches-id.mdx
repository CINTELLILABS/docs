---
title: "Get Test Batch"
api: "GET https://api.bland.ai/v1/agent-testing/batches/{id}"
description: "Retrieve a test batch and all its runs."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The batch ID (UUID v4).
</ParamField>

### Response

Returns `404 Not Found` if the batch does not exist or belongs to another organization.

<ResponseField name="data" type="object">
  <Expandable title="data object">
    <ResponseField name="id" type="string">
      The batch ID.
    </ResponseField>
    <ResponseField name="status" type="string">
      One of `PENDING`, `RUNNING`, `PASSED`, `FAILED`, `ERROR`, `CANCELLED`.
    </ResponseField>
    <ResponseField name="pathway_id" type="string">
      The pathway under test, or `null`.
    </ResponseField>
    <ResponseField name="persona_id" type="string">
      The persona under test, or `null`.
    </ResponseField>
    <ResponseField name="agent_id" type="string">
      The agent under test, or `null`.
    </ResponseField>
    <ResponseField name="version_number" type="integer">
      The pathway version tested, or `null` for the current production version.
    </ResponseField>
    <ResponseField name="total_scenarios" type="integer">
      Number of scenarios in the batch.
    </ResponseField>
    <ResponseField name="passed_scenarios" type="integer">
      Number of runs that passed.
    </ResponseField>
    <ResponseField name="failed_scenarios" type="integer">
      Number of runs that failed.
    </ResponseField>
    <ResponseField name="trigger_source" type="string">
      How the batch was started. `api` for batches created through this API; other values include `simulation` and `tornado`.
    </ResponseField>
    <ResponseField name="triggered_by" type="string">
      The user or organization ID that started the batch.
    </ResponseField>
    <ResponseField name="webhook_url" type="string">
      The webhook URL supplied at creation, or `null`.
    </ResponseField>
    <ResponseField name="publish_on_pass" type="boolean">
      Whether the pathway version is promoted when every run passes. Always `false` for batches created through this API.
    </ResponseField>
    <ResponseField name="environment" type="string">
      Deployment environment for CI-triggered batches, or `null`.
    </ResponseField>
    <ResponseField name="simulation_set_id" type="string">
      The simulation set this batch belongs to, or `null`.
    </ResponseField>
    <ResponseField name="simulation_index" type="integer">
      1-based position within the simulation set, or `null`.
    </ResponseField>
    <ResponseField name="tornado_session_id" type="string">
      The tornado session this batch belongs to, or `null`.
    </ResponseField>
    <ResponseField name="tornado_iteration" type="integer">
      The tornado iteration that produced this batch, or `null`.
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Arbitrary metadata, or `null`.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp.
    </ResponseField>
    <ResponseField name="completed_at" type="string">
      ISO 8601 timestamp, or `null` while running.
    </ResponseField>
    <ResponseField name="runs" type="array">
      The batch's runs, oldest first. Each run has the same top-level fields as [Get Test Run](/api-v1/get/agent-testing-runs-id) plus a `scenario` object with `name` and `category`. Assertion results are not included here; fetch a run by ID for them.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array">
  `null` on success.
</ResponseField>
