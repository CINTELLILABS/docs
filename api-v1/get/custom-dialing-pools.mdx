---
title: "List Custom Dialing Pools"
api: "GET https://api.bland.ai/v1/custom-dialing-pools"
description: "Retrieve all custom dialing pools for your organization."
---

<Note>
  Custom dialing is only available on Enterprise plans.
</Note>

Retrieve a list of all custom dialing pools that belong to your organization.

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Response

<ResponseField name="data" type="array">
  Array of custom dialing pool objects.
  
  <Expandable title="pool object properties">
    <ResponseField name="id" type="string">
      Unique identifier for the pool.
    </ResponseField>
    
    <ResponseField name="owner_id" type="string">
      Organization ID that owns this pool.
    </ResponseField>
    
    <ResponseField name="phone_numbers" type="array">
      Array of phone numbers in the pool.
    </ResponseField>

    <ResponseField name="phone_numbers_count" type="number">
      Number of phone numbers in the pool.
    </ResponseField>
    
    <ResponseField name="encrypted_key" type="string">
      UUID of associated Twilio credentials (or null if none).
    </ResponseField>

    <ResponseField name="credentials_name" type="string | null">
      Twilio account SID associated with the credentials, or `null`.
    </ResponseField>
    
    <ResponseField name="created_at" type="string">
      Timestamp when the pool was created.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array | null">
  `null` on success. On failure, contains error objects.
  
  <Expandable title="error object properties">
    <ResponseField name="error" type="string">
      Error code indicating the type of error.
    </ResponseField>
    
    <ResponseField name="message" type="string">
      Human-readable error message.
    </ResponseField>
  </Expandable>
</ResponseField>
