---
title: "Get Test Run"
api: "GET https://api.bland.ai/v1/agent-testing/runs/{id}"
description: "Retrieve detailed results for a specific test run including chat history, assertion results, and scores."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The run ID (UUID v4).
</ParamField>

### Response

Returns `404 Not Found` if the run does not exist or belongs to another organization.

<ResponseField name="data" type="object">
  <Expandable title="data object">
    <ResponseField name="id" type="string">
      The run ID.
    </ResponseField>
    <ResponseField name="scenario_id" type="string">
      The scenario that was executed.
    </ResponseField>
    <ResponseField name="pathway_id" type="string">
      The pathway under test, or `null`.
    </ResponseField>
    <ResponseField name="persona_id" type="string">
      The persona under test, or `null`.
    </ResponseField>
    <ResponseField name="agent_id" type="string">
      The agent under test, or `null`.
    </ResponseField>
    <ResponseField name="version_number" type="integer">
      The pathway version tested, or `null` for the current production version.
    </ResponseField>
    <ResponseField name="status" type="string">
      One of `PENDING`, `RUNNING`, `PASSED`, `FAILED`, `ERROR`, `CANCELLED`. A run passes when every assertion with `is_required: true` passed.
    </ResponseField>
    <ResponseField name="triggered_by" type="string">
      The user or organization ID that started the run.
    </ResponseField>
    <ResponseField name="trigger_source" type="string">
      `manual` for single runs, `api` for batch runs, `simulation` or `tornado` for runs created by those features.
    </ResponseField>
    <ResponseField name="call_id" type="string">
      The simulated call ID, or `null` before execution starts.
    </ResponseField>
    <ResponseField name="chat_history" type="array">
      The conversation. Each entry has `role` (`user` for the simulated caller, `assistant` for your agent) and `content`.
    </ResponseField>
    <ResponseField name="variables_extracted" type="object">
      Variables the agent extracted during the run, keyed by name.
    </ResponseField>
    <ResponseField name="nodes_visited" type="array">
      Ordered pathway node IDs visited during the run.
    </ResponseField>
    <ResponseField name="current_node_id" type="string">
      The node where the conversation ended, or `null`.
    </ResponseField>
    <ResponseField name="turn_count" type="integer">
      Number of conversational turns.
    </ResponseField>
    <ResponseField name="duration_ms" type="integer">
      Execution time in milliseconds.
    </ResponseField>
    <ResponseField name="llm_eval_count" type="integer">
      Number of LLM-backed assertion evaluations performed.
    </ResponseField>
    <ResponseField name="overall_score" type="number">
      Weighted average of assertion scores, from 0 to 1. `null` until the run completes.
    </ResponseField>
    <ResponseField name="bland_tone_score" type="object">
      Present when the scenario has `bland_tone_enabled: true`, otherwise `null`. Contains the 0 to 1 scores `naturalness`, `conciseness`, `empathy`, `flow_preservation`, `back_channeling`, `topic_transitions`, and `overall_tone_score`, plus `personality_appropriate` (boolean), `issues` (array of strings), and `examples` (object with `good` and `needs_improvement` string arrays).
    </ResponseField>
    <ResponseField name="assertion_results" type="array">
      One entry per assertion, in assertion order.

      <Expandable title="assertion result object">
        <ResponseField name="id" type="string">
          The result ID.
        </ResponseField>
        <ResponseField name="assertion_id" type="string">
          The assertion evaluated.
        </ResponseField>
        <ResponseField name="status" type="string">
          `PASSED`, `FAILED`, or `ERROR`.
        </ResponseField>
        <ResponseField name="score" type="number">
          Assertion score from 0 to 1, or `null`.
        </ResponseField>
        <ResponseField name="value" type="any">
          The evaluated value (for example the extracted variable or the judge verdict), or `null`.
        </ResponseField>
        <ResponseField name="reasoning" type="string">
          Explanation from LLM-backed assertions, or `null`.
        </ResponseField>
        <ResponseField name="duration_ms" type="integer">
          Evaluation time in milliseconds.
        </ResponseField>
        <ResponseField name="error_message" type="string">
          Set when `status` is `ERROR`, otherwise `null`.
        </ResponseField>
        <ResponseField name="assertion" type="object">
          The assertion definition: `name`, `type`, `is_required`, `weight`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="scenario" type="object">
      The scenario's `name`, `category`, `bland_tone_enabled`, and `tester_persona_name`.
    </ResponseField>
    <ResponseField name="error_message" type="string">
      Set when `status` is `ERROR`, otherwise `null`.
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Run metadata. After [Analyze Failed Run](/api-v1/post/agent-testing-runs-analyze) it contains `norm_analysis`.
    </ResponseField>
    <ResponseField name="batch_id" type="string">
      The batch this run belongs to, or `null`.
    </ResponseField>
    <ResponseField name="simulation_set_id" type="string">
      The simulation set this run belongs to, or `null`.
    </ResponseField>
    <ResponseField name="simulation_index" type="integer">
      1-based position within the simulation set, or `null`.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp.
    </ResponseField>
    <ResponseField name="completed_at" type="string">
      ISO 8601 timestamp, or `null` until the run finishes.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array">
  `null` on success.
</ResponseField>
