---
title: "Get Test Scenario"
api: "GET https://api.bland.ai/v1/agent-testing/scenarios/{id}"
description: "Retrieve a specific test scenario with its assertions."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The scenario ID (UUID v4).
</ParamField>

### Response

Returns `404 Not Found` if the scenario does not exist or belongs to another organization.

<ResponseField name="data" type="object">
  <Expandable title="data object">
    <ResponseField name="id" type="string">
      The scenario ID.
    </ResponseField>
    <ResponseField name="org_id" type="string">
      Organization that owns the scenario.
    </ResponseField>
    <ResponseField name="pathway_id" type="string">
      The pathway under test, or `null`.
    </ResponseField>
    <ResponseField name="persona_id" type="string">
      The persona under test, or `null`.
    </ResponseField>
    <ResponseField name="agent_id" type="string">
      The agent under test, or `null`. Exactly one of `pathway_id`, `persona_id`, `agent_id` is set.
    </ResponseField>
    <ResponseField name="name" type="string">
      Scenario name.
    </ResponseField>
    <ResponseField name="description" type="string">
      Scenario description, or `null`.
    </ResponseField>
    <ResponseField name="category" type="string">
      One of `CUSTOM`, `VOICEMAIL`, `VOICEMAIL_SCREENER`, `ANGRY_CALLER`, `BELLIGERENT_CALLER`, `CONFUSED_CALLER`, `CALL_SCREENER`, `HAPPY_PATH`, `EDGE_CASE`.
    </ResponseField>
    <ResponseField name="scenario_type" type="string">
      One of `AGENT`, `REPLAY`, `HISTORICAL`.
    </ResponseField>
    <ResponseField name="tester_persona_prompt" type="string">
      Prompt for the simulated caller, or `null`.
    </ResponseField>
    <ResponseField name="tester_persona_name" type="string">
      Display name for the simulated caller, or `null`.
    </ResponseField>
    <ResponseField name="max_turns" type="integer">
      Maximum conversation turns.
    </ResponseField>
    <ResponseField name="request_data" type="object">
      Request data passed to the agent, or `null`.
    </ResponseField>
    <ResponseField name="start_node_id" type="string">
      Starting node ID, or `null`.
    </ResponseField>
    <ResponseField name="bland_tone_enabled" type="boolean">
      Whether Bland Tone scoring runs.
    </ResponseField>
    <ResponseField name="is_template" type="boolean">
      Always `false` for scenarios you own. Templates are listed by [List Test Templates](/api-v1/get/agent-testing-templates).
    </ResponseField>
    <ResponseField name="template_category" type="string">
      Reserved for templates, `null` otherwise.
    </ResponseField>
    <ResponseField name="enabled" type="boolean">
      Whether the scenario is enabled.
    </ResponseField>
    <ResponseField name="is_required_for_promotion" type="boolean">
      Whether the scenario must pass before the pathway can be promoted.
    </ResponseField>
    <ResponseField name="input_messages" type="array">
      Pre-seeded caller messages for `REPLAY` scenarios. Empty array when unused.
    </ResponseField>
    <ResponseField name="advanced_instructions" type="string">
      Additional execution instructions, or `null`.
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Arbitrary metadata, or `null`.
    </ResponseField>
    <ResponseField name="tags" type="array">
      String tags. Empty array when unused.
    </ResponseField>
    <ResponseField name="assertions" type="array">
      Assertions in `order` ascending. Each has `id`, `scenario_id`, `type`, `config`, `name`, `is_required`, `weight`, `order`, `created_at`.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp.
    </ResponseField>
    <ResponseField name="updated_at" type="string">
      ISO 8601 timestamp.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array">
  `null` on success.
</ResponseField>
