---
title: "Get Simulation Set"
api: "GET https://api.bland.ai/v1/agent-testing/simulation-sets/{id}"
description: "Retrieve a simulation set with its statistics including pass rates, flakiness detection, and score distributions."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The simulation set ID (UUID v4).
</ParamField>

### Response

Returns `404 Not Found` if the set does not exist or belongs to another organization.

<ResponseField name="data" type="object">
  <Expandable title="data object">
    <ResponseField name="id" type="string">
      The simulation set ID.
    </ResponseField>
    <ResponseField name="status" type="string">
      One of `PENDING`, `RUNNING`, `PASSED`, `FAILED`, `ERROR`, `CANCELLED`. `PASSED` means every run passed; `FAILED` means at least one run failed.
    </ResponseField>
    <ResponseField name="pathway_id" type="string">
      The pathway under test, or `null`.
    </ResponseField>
    <ResponseField name="persona_id" type="string">
      The persona under test, or `null`.
    </ResponseField>
    <ResponseField name="agent_id" type="string">
      The agent under test, or `null`.
    </ResponseField>
    <ResponseField name="version_number" type="integer">
      The pathway version tested, or `null`.
    </ResponseField>
    <ResponseField name="agent_version_id" type="string">
      The agent version pinned for the set, or `null` for the latest version.
    </ResponseField>
    <ResponseField name="triggered_by" type="string">
      The user or organization ID that created the set.
    </ResponseField>
    <ResponseField name="simulations_per_scenario" type="integer">
      Number of times each scenario runs.
    </ResponseField>
    <ResponseField name="total_scenarios" type="integer">
      Number of scenarios in the set.
    </ResponseField>
    <ResponseField name="statistics" type="object">
      `null` until the set completes.

      <Expandable title="statistics object">
        <ResponseField name="per_scenario" type="object">
          Map of scenario ID to `scenario_id`, `scenario_name`, `total_runs`, `passed`, `failed`, `pass_rate` (0 to 1), `is_flaky` (some runs passed and some failed), `scores` (`mean`, `median`, `stddev`, `min`, `max`, each 0 to 1), `worst_run_id`, `common_failure_modes` (array of strings), and `confidence` (`low`, `medium`, or `high`).
        </ResponseField>
        <ResponseField name="overall" type="object">
          `avg_pass_rate`, `flaky_count`, `reliable_count`, `total_runs`.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Arbitrary metadata, or `null`.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 timestamp.
    </ResponseField>
    <ResponseField name="completed_at" type="string">
      ISO 8601 timestamp, or `null` while running.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array">
  `null` on success.
</ResponseField>
