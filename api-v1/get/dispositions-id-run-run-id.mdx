---
title: "Get Outcome Run"
api: "GET https://api.bland.ai/v1/dispositions/{id}/run/{run_id}"
description: "Get one saved run of an outcome."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The outcome ID.
</ParamField>

<ParamField path="run_id" type="string" required>
  The run ID, from [List Outcome Runs](/api-v1/get/dispositions-id-run).
</ParamField>

### Response

<ResponseField name="data" type="object">
  The run.

  <Expandable title="Run fields">
    <ResponseField name="id" type="string">
      The run ID.
    </ResponseField>

    <ResponseField name="org_id" type="string">
      The organization that owns the run.
    </ResponseField>

    <ResponseField name="disposition_id" type="string">
      The outcome that ran.
    </ResponseField>

    <ResponseField name="conversation_id" type="string">
      The call or SMS conversation the outcome ran on.
    </ResponseField>

    <ResponseField name="reference_type" type="string">
      `CALL`, `SMS` or `CALL_TEST`.
    </ResponseField>

    <ResponseField name="snippet_version" type="integer | null">
      The script version that ran.
    </ResponseField>

    <ResponseField name="status" type="string">
      `PENDING`, `COMPLETE` or `FAILED`.
    </ResponseField>

    <ResponseField name="result" type="any">
      What the script returned. `null` until the run completes, and on a failed run.
    </ResponseField>

    <ResponseField name="console" type="object | null">
      Console output the script logged, when captured.
    </ResponseField>

    <ResponseField name="error" type="string | null">
      Why the run failed. `null` when it succeeded.
    </ResponseField>

    <ResponseField name="exec_time" type="integer | null">
      How long the script ran, in milliseconds.
    </ResponseField>

    <ResponseField name="input" type="object | null">
      The conversation data the script received.
    </ResponseField>

    <ResponseField name="created_at" type="string">
      When the run started (ISO 8601).
    </ResponseField>

    <ResponseField name="finished_at" type="string | null">
      When the run finished (ISO 8601).
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="null | array">
  `null` on success, otherwise an array of error objects.
</ResponseField>
