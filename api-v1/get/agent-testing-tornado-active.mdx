---
title: "Get Active Tornado Session"
api: "GET https://api.bland.ai/v1/agent-testing/tornado/active"
description: "Get the newest active tornado session for a pathway or persona."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Query Parameters

<ParamField query="pathway_id" type="string">
  Filter by pathway. Provide `pathway_id`, `persona_id`, or both.
</ParamField>

<ParamField query="persona_id" type="string">
  Filter by persona. Provide `pathway_id`, `persona_id`, or both.
</ParamField>

### Response

Returns the newest matching `RUNNING` session as the raw stored session row.

<ResponseField name="data" type="object">
  <Expandable title="Session fields">
    <ResponseField name="id" type="string">Session UUID.</ResponseField>
    <ResponseField name="created_at" type="string" />
    <ResponseField name="updated_at" type="string" />
    <ResponseField name="completed_at" type="string | null" />
    <ResponseField name="org_id" type="string" />
    <ResponseField name="pathway_id" type="string | null" />
    <ResponseField name="persona_id" type="string | null" />
    <ResponseField name="status" type="string">`RUNNING`.</ResponseField>
    <ResponseField name="triggered_by" type="string | null" />
    <ResponseField name="max_iterations" type="integer" />
    <ResponseField name="timeout_ms" type="integer" />
    <ResponseField name="scenario_ids" type="string[]" />
    <ResponseField name="current_iteration" type="integer" />
    <ResponseField name="iterations_data" type="object | null" />
    <ResponseField name="total_scenarios" type="integer" />
    <ResponseField name="passed_scenarios" type="integer" />
    <ResponseField name="failed_scenarios" type="integer" />
    <ResponseField name="unfixable_ids" type="string[]" />
    <ResponseField name="norm_session_id" type="string | null" />
    <ResponseField name="forked_version" type="integer | null" />
    <ResponseField name="error_message" type="string | null" />
    <ResponseField name="metadata" type="object | null" />
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="null | array">
  `null` on success. Returns `400` if neither filter is provided and `404` if no active session matches.
</ResponseField>
