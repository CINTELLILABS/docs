---
title: "Get Production Persona Version"
api: "GET https://api.bland.ai/v1/personas/{persona_id}/production"
description: "Retrieve the current production version of a persona."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="persona_id" type="string" required>
  The unique identifier of the persona.
</ParamField>

### Response

<ResponseField name="data" type="object | null">
  The current production version. This is `null` when the persona does not have a production version.

  <Expandable title="version object">
    <ResponseField name="id" type="string">
      The version ID.
    </ResponseField>
    <ResponseField name="persona_id" type="string">
      The persona ID.
    </ResponseField>
    <ResponseField name="version_type" type="string">
      `production` for the current production version.
    </ResponseField>
    <ResponseField name="version_number" type="number">
      The sequential version number.
    </ResponseField>
    <ResponseField name="orchestration_prompt" type="string | null">
      Instructions for how the persona handles a conversation.
    </ResponseField>
    <ResponseField name="personality_prompt" type="string | null">
      Instructions for the persona's tone and behavior.
    </ResponseField>
    <ResponseField name="pathway_conditions" type="object | null">
      Conditions that control pathway behavior.
    </ResponseField>
    <ResponseField name="kb_ids" type="array">
      Knowledge base IDs available to the version.
    </ResponseField>
    <ResponseField name="call_config" type="object | null">
      Call settings accepted by the [Send Call endpoint](/api-v1/post/calls).
    </ResponseField>
    <ResponseField name="default_tools" type="array">
      Tools available to the version.
    </ResponseField>
    <ResponseField name="memory_enabled" type="boolean">
      Whether memory is enabled.
    </ResponseField>
    <ResponseField name="entity_schemas" type="object | null">
      Entity schemas used by memory extraction.
    </ResponseField>
    <ResponseField name="auth_config" type="object | null">
      Authentication configuration for the version.
    </ResponseField>
    <ResponseField name="skills" type="object | null">
      Skills configured for the version.
    </ResponseField>
    <ResponseField name="promoted_from_version_id" type="string | null">
      The source version ID used during promotion.
    </ResponseField>
    <ResponseField name="promoted_at" type="string | null">
      ISO 8601 timestamp for the promotion.
    </ResponseField>
    <ResponseField name="promoted_by" type="string | null">
      The user ID that promoted the version.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 creation timestamp.
    </ResponseField>
    <ResponseField name="updated_at" type="string">
      ISO 8601 update timestamp.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array | null">
  `null` on success. An unknown persona returns `404 PERSONA_NOT_FOUND`.
</ResponseField>
