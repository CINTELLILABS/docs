---
title: "Get Enhanced Analytics"
api: "GET https://api.bland.ai/v1/agent-testing/analytics/{pathwayId}/enhanced"
description: "Get enhanced testing analytics including node failure heatmap, weakest link, trend analysis, and Sankey flow data."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Path Parameters

<ParamField path="pathwayId" type="string" required>
  The pathway ID (UUID v4).
</ParamField>

### Query Parameters

<ParamField query="days" type="integer" default="30">
  Number of days to look back. Clamped to the range 1 to 365.
</ParamField>

### Response

Analytics are computed from the 500 most recent runs in the window, across every status. `failed` counts runs in `FAILED` or `ERROR`.

<ResponseField name="data" type="object">
  <Expandable title="data object">
    <ResponseField name="total_runs" type="integer">
      Number of test runs in the time window.
    </ResponseField>
    <ResponseField name="passed" type="integer">
      Number of test runs with status `PASSED`.
    </ResponseField>
    <ResponseField name="failed" type="integer">
      Number of test runs with status `FAILED` or `ERROR`.
    </ResponseField>
    <ResponseField name="pass_rate" type="number">
      Pass rate as a decimal from 0 to 1.
    </ResponseField>
    <ResponseField name="avg_score" type="number">
      Average `overall_score` across all runs, from 0 to 1.
    </ResponseField>
    <ResponseField name="health_score" type="number">
      Overall health score from 0 to 1. Weighted blend of pass rate (50%), average score (20%), trend direction (15%), and non-flakiness (15%).
    </ResponseField>
    <ResponseField name="reliability_score" type="integer">
      Consistency score from 0 to 100: `(1 - flaky_rate) * pass_rate * 100`, rounded.
    </ResponseField>
    <ResponseField name="trend" type="object">
      <Expandable title="trend object">
        <ResponseField name="weekly" type="string">
          One of `improving`, `declining`, `stable`. Always `stable` with fewer than 3 days of data.
        </ResponseField>
        <ResponseField name="daily_pass_rates" type="array">
          One entry per day, oldest first: `date` (`YYYY-MM-DD`), `rate` (0 to 1), and `count` (runs that day).
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="node_failure_heatmap" type="array">
      One entry per pathway node visited, sorted by `failure_rate` descending.

      <Expandable title="node failure object">
        <ResponseField name="node_id" type="string">
          The node ID.
        </ResponseField>
        <ResponseField name="node_name" type="string">
          The node's display name, or its ID when the name is unavailable.
        </ResponseField>
        <ResponseField name="failure_count" type="integer">
          Number of failed or errored runs that traversed this node.
        </ResponseField>
        <ResponseField name="total_traversals" type="integer">
          Number of runs that traversed this node.
        </ResponseField>
        <ResponseField name="failure_rate" type="number">
          `failure_count / total_traversals`, from 0 to 1.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="weakest_link" type="object">
      The first heatmap entry (`node_id`, `node_name`, `failure_rate`), or `null` when no runs visited any node.
    </ResponseField>
    <ResponseField name="sankey_data" type="object">
      <Expandable title="sankey data object">
        <ResponseField name="nodes" type="array">
          Flow nodes. Each has `id`, `name`, and `type` (one of `scenario`, `pathway_node`, `outcome`). Scenario nodes group runs by scenario category and carry a `category` field. Outcome nodes have IDs `outcome-PASSED` and `outcome-FAILED`.
        </ResponseField>
        <ResponseField name="links" type="array">
          Edges between nodes: `source`, `target`, `value` (traversals), `pass_count`, and `fail_count`.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array">
  `null` on success.
</ResponseField>
