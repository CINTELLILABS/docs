{/*
  The colour engine.

  Positions are designed and fixed: the L, the band across the top, the way the
  field thins as it travels. Colour is systematic, so it is generated, and the
  field is a different one on every load.

  Two rules stop "random" from looking worse than a hand-built field, both of
  them measured off the 88 cells that were hand-built before this:

  1. WEIGHTED, not uniform. Orange and Yellow were 45% of the field between
     them; the four darks together were 13%. Uniform picking across twelve
     colours would have pushed the darks to 33% and the page would read heavy
     and muddled. The weights below ARE the hand-built mix, counted.

  2. NO TOUCHING TWINS. The hand-built field had 2 adjacent same-colour pairs in
     88 cells. Uniform picking averages 7 to 14, and each one welds two 72px
     modules into a single 144px block, which is the one thing that destroys the
     deck grid. A colour is rejected if the cell to its left or above already
     has it.

  SEED is null, so the field is new on every load. Set it to a number to freeze
  one, which is what to do before taking a screenshot or comparing two builds.
*/}
export const PALETTE = [
  ['#FF5900', 20],  // Orange
  ['#F9A900', 20],  // Yellow
  ['#048244',  9],  // Jade
  ['#C238DA',  8],  // Violet
  ['#1C54A5',  8],  // Blue
  ['#CA071A',  5],  // Rotary Red
  ['#001847',  4],  // Indigo
  ['#5E007E',  4],  // Purple
  ['#DB005F',  3],  // Magenta
  ['#151515',  3],  // Off-Black
  ['#FF4294',  3],  // Rose
  ['#004121',  1],  // Green
];

export const SEED = null;

export const WEIGHT_TOTAL = PALETTE.reduce((n, p) => n + p[1], 0);

{/* Small LCG so a seed reproduces a field exactly. Math.imul keeps it in 32 bits. */}
export const makeRandom = (seed) => {
  let x = (seed == null ? Math.floor(Math.random() * 4294967296) : seed) >>> 0;
  return () => {
    x = (Math.imul(x, 1664525) + 1013904223) >>> 0;
    return x / 4294967296;
  };
};

export const pick = (rnd, banned) => {
  for (let tries = 0; tries < 24; tries++) {
    let t = rnd() * WEIGHT_TOTAL;
    let chosen = PALETTE[PALETTE.length - 1][0];
    for (const [hex, w] of PALETTE) {
      t -= w;
      if (t <= 0) { chosen = hex; break; }
    }
    if (!banned.includes(chosen)) return chosen;
  }
  return PALETTE.map(p => p[0]).find(h => !banned.includes(h));
};

{/* Paint a set of positions, refusing a colour its left or upper neighbour holds.

    Sorted into reading order first, which is load-bearing rather than tidy: the
    rejection can only see neighbours already placed, and the authored position
    list runs columns 1 to 5 down the L before it runs the band across, so
    painting in array order leaves some left neighbours unpainted and lets a
    twin through. Measured: 1 adjacent pair slipped past on a sample load before
    this sort, 0 after. */}
export const paint = (positions, rnd) => {
  const placed = new Map();
  const ordered = [...positions].sort((a, b) => (a[1] - b[1]) || (a[0] - b[0]));
  return ordered.map(([col, row]) => {
    const banned = [placed.get((col - 1) + ':' + row), placed.get(col + ':' + (row - 1))].filter(Boolean);
    const c = pick(rnd, banned);
    placed.set(col + ':' + row, c);
    return { c, col, row, alt: '', dur: 0, delay: 0 };
  });
};

{/* A few cells carry a second colour underneath and change slowly. Which ones
    is part of what varies, so the field is never quite the same twice. */}
export const withBreath = (cells, rnd, count) => {
  const chosen = new Set();
  while (chosen.size < count && chosen.size < cells.length) chosen.add(Math.floor(rnd() * cells.length));
  return cells.map((cell, i) => chosen.has(i)
    ? { ...cell, alt: pick(rnd, [cell.c]), dur: 19 + Math.floor(rnd() * 10), delay: Math.floor(rnd() * 12) }
    : cell);
};

{/*
  The settle sweep.

  A cell's delay is its own diagonal (col + row), scaled so the last diagonal in
  the field lands at SETTLE_WINDOW. The field therefore assembles outward from
  the anchor corner in one continuous pass, and it takes the same time to do it
  whatever the field's width.

  Scaling rather than clamping is what makes it width-independent. A fixed step
  with a ceiling works only while the field is narrow enough never to reach the
  ceiling. Widen the grid and every cell past that point pins to the same delay
  and arrives on one frame, so the left of the field sweeps in and the right
  half appears in a block.

  MAX_DIAGONAL is read off the data, not written down, so adding columns moves
  the sweep with them instead of reintroducing this.
*/}
export const SETTLE_WINDOW = 720;
export const MAX_DIAGONAL = Math.max(...CELLS.map(c => c.col + c.row));

{/*
  One module. Colour and grid line are inline style so none of it goes through
  Tailwind. Scatter cells sit below the field's own diagonals, so they land
  after it, which is the right order.
*/}
export const Block = ({ c, col, row, alt, dur, delay }) => (
  <div
    className="splash-cell splash-settle relative"
    style={{
      backgroundColor: alt ? alt : c,
      gridColumn: String(col),
      gridRow: String(row),
      animationDelay: Math.round((col + row) / MAX_DIAGONAL * SETTLE_WINDOW) + 'ms'
    }}
  >
    {alt ? (
      <span
        className="splash-breathe absolute inset-0 block"
        style={{
          backgroundColor: c,
          '--splash-breathe-dur': dur + 's',
          '--splash-breathe-delay': delay + 's'
        }}
      />
    ) : null}
  </div>
);
