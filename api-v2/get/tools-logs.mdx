---
title: "Tool Execution Logs"
api: "GET https://api.bland.ai/v2/tools/logs"
description: "Retrieve per-execution logs for your tools with optional filtering and pagination."
---

### Headers

<ParamField header="authorization" type="string" required>
  Your API key for authentication.
</ParamField>

### Query Parameters

**Pagination**

<ParamField query="page" type="integer">
  Page number. Defaults to `1`.
</ParamField>

<ParamField query="pageSize" type="integer">
  Results per page. Maximum `100`. Defaults to `50`.
</ParamField>

**Filters**

<ParamField query="tool_id" type="string">
  Filter logs by a specific tool UUID (without the `TL-` prefix).
</ParamField>

<ParamField query="resource_id" type="string">
  Filter logs by a specific resource ID.
</ParamField>

<ParamField query="integration" type="string">
  Filter logs by integration key (e.g. `"slack"`).
</ParamField>

<ParamField query="action" type="string">
  Filter logs by action name.
</ParamField>

<ParamField query="status" type="string">
  Filter by execution status. Allowed values: `"success"`, `"error"`.
</ParamField>

<ParamField query="call_id" type="string">
  Filter logs by the call ID that triggered the tool execution.
</ParamField>

<ParamField query="error_type" type="string">
  Filter logs by error type (only applies to failed executions).
</ParamField>

**Date Range**

<ParamField query="created_after" type="string">
  ISO 8601 date string. Only return logs created after this timestamp.
</ParamField>

<ParamField query="created_before" type="string">
  ISO 8601 date string. Only return logs created before this timestamp.
</ParamField>

**Size Filters**

A value of `0` is treated as unset. Query parameters not listed here are ignored.

<ParamField query="execution_time_ms_gt" type="number">
  Only return logs where execution time is greater than this value (in ms).
</ParamField>

<ParamField query="execution_time_ms_lt" type="number">
  Only return logs where execution time is less than this value (in ms).
</ParamField>

<ParamField query="input_size_kb_gt" type="number">
  Only return logs where input payload size is greater than this value (in KB).
</ParamField>

<ParamField query="input_size_kb_lt" type="number">
  Only return logs where input payload size is less than this value (in KB).
</ParamField>

<ParamField query="output_size_kb_gt" type="number">
  Only return logs where output payload size is greater than this value (in KB).
</ParamField>

<ParamField query="output_size_kb_lt" type="number">
  Only return logs where output payload size is less than this value (in KB).
</ParamField>

### Response

<ResponseField name="data" type="object">
  The page of logs and its pagination counters.

  <Expandable title="data">
    <ResponseField name="logs" type="array">
      Array of tool execution log entries, newest first.

      <Expandable title="log entry">
        <ResponseField name="id" type="string">
          Unique log entry identifier (UUID).
        </ResponseField>
        <ResponseField name="tool_id" type="string | null">
          UUID of the tool that was executed (no `TL-` prefix). `null` if the tool has since been deleted.
        </ResponseField>
        <ResponseField name="call_id" type="string | null">
          ID of the call that triggered this execution.
        </ResponseField>
        <ResponseField name="resource_id" type="string | null">
          ID of the resource used, if any.
        </ResponseField>
        <ResponseField name="integration" type="string">
          Integration key that was invoked.
        </ResponseField>
        <ResponseField name="action" type="string">
          Action that was invoked.
        </ResponseField>
        <ResponseField name="inputs" type="object">
          The inputs the tool was executed with. This can include values the caller said on the call, so treat the field as sensitive.
        </ResponseField>
        <ResponseField name="outputs" type="object | null">
          The integration's response payload, if one was recorded.
        </ResponseField>
        <ResponseField name="status" type="string">
          Execution result: `"success"` or `"error"`.
        </ResponseField>
        <ResponseField name="error_message" type="string | null">
          Error text if the execution failed.
        </ResponseField>
        <ResponseField name="error_type" type="string | null">
          Error category if the execution failed.
        </ResponseField>
        <ResponseField name="execution_time_ms" type="number">
          Time taken to execute the tool in milliseconds.
        </ResponseField>
        <ResponseField name="input_size_kb" type="number">
          Size of the input payload in KB.
        </ResponseField>
        <ResponseField name="output_size_kb" type="number | null">
          Size of the output payload in KB.
        </ResponseField>
        <ResponseField name="user_id" type="string">
          UUID of the user the execution ran as.
        </ResponseField>
        <ResponseField name="org_id" type="string">
          UUID of your organization.
        </ResponseField>
        <ResponseField name="created_at" type="string">
          ISO 8601 timestamp of when the execution occurred.
        </ResponseField>
        <ResponseField name="updated_at" type="string">
          ISO 8601 timestamp of the last update to the entry.
        </ResponseField>
        <ResponseField name="created_date" type="string">
          Calendar date of the execution (UTC), used for partitioning.
        </ResponseField>
        <ResponseField name="resource" type="object | null">
          The linked resource: `id`, `name`, `integration`, `status`, `is_active`.
        </ResponseField>
        <ResponseField name="tool" type="object | null">
          The linked tool: `id`, `label`, and its `tool` definition.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="totalLogs" type="number">
      Total number of logs matching the filters.
    </ResponseField>

    <ResponseField name="totalPages" type="number">
      Total number of pages.
    </ResponseField>

    <ResponseField name="currentPage" type="number">
      The current page number.
    </ResponseField>

    <ResponseField name="pageSize" type="number">
      Number of results per page.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="errors" type="array | null">
  An empty array on success, or a list of `{ error, message }` objects if the request failed.
</ResponseField>

### Errors

| Status | `error` | When |
|---|---|---|
| 400 | `INVALID_PARAMETER` | A query parameter has the wrong type or is out of range (for example `pageSize` above 100). |
| 500 | `INTERNAL_SERVER_ERROR` | Logs could not be retrieved. |
